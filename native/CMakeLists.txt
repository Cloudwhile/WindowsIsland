cmake_minimum_required(VERSION 3.21)
project(WindowsIslandTelegramHook LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
    vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC vendor/minhook/include)
add_library(WindowsIsland.TelegramHook SHARED telegram/bridge.cpp telegram/capture.cpp telegram/profile.cpp telegram/avatar.cpp telegram/events.cpp)
target_link_libraries(WindowsIsland.TelegramHook PRIVATE minhook)
target_compile_definitions(WindowsIsland.TelegramHook PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(WindowsIsland.TelegramHook PRIVATE /W4 /utf-8 /EHsc)
add_executable(WindowsIsland.TelegramFixture tests/fixture.cpp)
target_compile_definitions(WindowsIsland.TelegramFixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(WindowsIsland.TelegramFixture PRIVATE /W4 /utf-8 /EHsc /Od)
